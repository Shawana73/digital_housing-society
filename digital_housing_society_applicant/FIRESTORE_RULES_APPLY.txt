rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }

    function isOwner(uid) {
      return signedIn() && request.auth.uid == uid;
    }

    function ownsIncomingApplicant() {
      return signedIn() && request.resource.data.applicantId == request.auth.uid;
    }

    function ownsExistingApplicant() {
      return signedIn() && resource.data.applicantId == request.auth.uid;
    }

    function ownsIncomingRecipient() {
      return signedIn() && request.resource.data.recipientId == request.auth.uid;
    }

    function ownsExistingRecipient() {
      return signedIn() && resource.data.recipientId == request.auth.uid;
    }

    match /applicants/{uid} {
      allow create, read, update: if isOwner(uid);
      allow delete: if false;
    }

    match /applications/{applicationId} {
      allow create: if ownsIncomingApplicant();
      allow read, update, delete: if ownsExistingApplicant();
    }

    match /uploads/{uploadId} {
      allow create, update: if ownsIncomingApplicant() || isOwner(uploadId);
      allow read, delete: if ownsExistingApplicant() || isOwner(uploadId);
    }

    match /payments/{paymentId} {
      allow create, update: if ownsIncomingApplicant() || isOwner(paymentId);
      allow read, delete: if ownsExistingApplicant() || isOwner(paymentId);
    }

    match /notifications/{notificationId} {
      allow create: if signedIn();
      allow read, update, delete: if ownsExistingRecipient();
    }

    match /contacts/{contactId} {
      allow create: if signedIn();
      allow read, update: if ownsExistingApplicant();
      allow delete: if false;
    }

    match /plots/{plotId} {
      allow read: if signedIn();
      allow create, update, delete: if false;
    }

    match /ballot_results/{resultId} {
      allow read: if signedIn();
      allow create, update, delete: if false;
    }

    match /ballot_config/{docId} {
      allow read: if signedIn();
      allow create, update: if signedIn();
      allow delete: if false;
    }

    match /ballot_updates/{docId} {
      allow read: if signedIn();
      allow create, update: if signedIn();
      allow delete: if false;
    }

    match /ballot_live_results/{docId} {
      allow read: if signedIn();
      allow create, update: if signedIn();
      allow delete: if false;
    }

    match /payment_config/{docId} {
      allow read: if signedIn();
      allow create, update: if signedIn();
      allow delete: if false;
    }

    match /app_metadata/{docId} {
      allow read, create, update: if signedIn();
      allow delete: if false;
    }

    match /signup_otps/{email} {
      allow read, create, update: if true;
      allow delete: if false;
    }
  }
}
